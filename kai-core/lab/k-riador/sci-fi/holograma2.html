function drawLuva(landmarks, estado) { const cx = landmarks[9].x * canvasElement.width; const cy = landmarks[9].y * canvasElement.height; // Configuração de Estilo "4D" canvasCtx.shadowBlur = 20; canvasCtx.shadowColor = (estado === 'tiro') ? '#ff0000' : '#00ffff'; canvasCtx.lineWidth = 4; canvasCtx.strokeStyle = (estado === 'tiro') ? '#ff0000' : '#00ffff'; // 1. DESENHO DA ESTRUTURA (MESH) window.drawConnectors(canvasCtx, landmarks, HAND_CONNECTIONS, { color: (estado === 'tiro') ? '#ff4444' : '#00ffff', lineWidth: 2 }); // 2. EFEITOS DE ESTADO (VISUALIZAÇÃO 4D) if (estado === 'escudo') { // Campo de Força Esférico canvasCtx.beginPath(); canvasCtx.arc(cx, cy, 70, 0, Math.PI * 2); canvasCtx.fillStyle = 'rgba(0, 255, 255, 0.15)'; canvasCtx.fill(); canvasCtx.stroke(); } else if (estado === 'tiro') { // Feixe de Plasma const ponta = landmarks[8]; // Indicador canvasCtx.beginPath(); canvasCtx.moveTo(cx, cy); canvasCtx.lineTo(ponta.x * canvasElement.width, ponta.y * canvasElement.height); canvasCtx.strokeStyle = '#ff0000'; canvasCtx.lineWidth = 15; // Feixe grosso canvasCtx.stroke(); } else if (estado === 'scanner') { // Ondas de Pulso (Concentric Rings) const time = Date.now() * 0.005; for(let i=0; i<3; i++) { let radius = (40 + (i * 20)) + (Math.sin(time) * 10); canvasCtx.beginPath(); canvasCtx.arc(cx, cy, radius, 0, Math.PI * 2); canvasCtx.strokeStyle = `rgba(0, 255, 255, ${0.6 - i*0.2})`; canvasCtx.stroke(); } } // 3. NÚCLEO (PALMA) canvasCtx.beginPath(); canvasCtx.arc(cx, cy, 10, 0, Math.PI * 2); canvasCtx.fillStyle = (estado === 'tiro') ? '#ff0000' : '#ffffff'; canvasCtx.fill(); }
