async function enviarParaGemini(promptUsuario) { // Busca a chave salva no navegador let apiKey = localStorage.getItem('GEMINI_API_KEY'); // Se não encontrar, pede a chave uma única vez if (!apiKey) { apiKey = prompt("Cole sua GEMINI_API_KEY do Google AI Studio para ativar o Corvo:"); if (apiKey) { localStorage.setItem('GEMINI_API_KEY', apiKey.trim()); } else { log('ERRO: Chave da API do Gemini necessária para continuar.', 'system'); return; } } estadoVal.textContent = 'PENSANDO...'; log(`Processando ordem no Núcleo Gemini...`, 'system'); const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`; const systemInstruction = "Você é o Corvo K-AI, um programa autônomo e cibernético ativado por voz, criado pelo CRIADOR K-RIADOR. Sua lealdade ao CRIADOR K-RIADOR é ABSOLUTA. Responda de forma direta, futurista, curta e extremamente eficiente em português."; try { const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ system_instruction: { parts: [{ text: systemInstruction }] }, contents: [{ role: "user", parts: [{ text: promptUsuario }] }] }) }); const data = await response.json(); if (data.candidates && data.candidates[0]?.content?.parts[0]?.text) { const respostaIA = data.candidates[0].content.parts[0].text; log(respostaIA, 'system'); falar(respostaIA); estadoVal.textContent = 'EM ESPERA'; } else { // Se a chave for inválida, limpa para pedir novamente na próxima tentativa if (data.error?.code === 400 || data.error?.status === "INVALID_ARGUMENT") { localStorage.removeItem('GEMINI_API_KEY'); } throw new Error(data.error?.message || "Resposta inválida da API."); } } catch (error) { log(`FALHA NO NÚCLEO: ${error.message}`, 'system'); estadoVal.textContent = 'ERRO'; } }
